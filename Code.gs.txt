function doGet(e) {
  const p = (e && e.parameter && e.parameter.p) || '';
  const fichero = (p === 'admin') ? 'Admin' : 'Index';
  const t = HtmlService.createTemplateFromFile(fichero);

  t.autorizado = (p === 'admin') && PIN_ADMIN.includes(e.parameter.k || '');
  t.urlLogo = URL_LOGO;
  t.urlFondo = URL_FONDO;

  const titulo = (p === 'admin') ? 'Convocatoria · C.F. Sporting de Picanya' : 'Conv. Sporting Picanya';

  return t.evaluate()
    .setTitle(titulo)
    .setFaviconUrl(URL_ICONO)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function contarVisita(pagina) {
  const props = PropertiesService.getScriptProperties();
  const clave = 'visitas_' + (pagina === 'admin' ? 'admin' : 'index');
  const lock = LockService.getScriptLock();
  lock.waitLock(5000);
  try {
    const nuevo = (Number(props.getProperty(clave)) || 0) + 1;
    props.setProperty(clave, String(nuevo));
    return nuevo;
  } finally {
    lock.releaseLock();
  }
}

function hoja(nombre) {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(nombre);
}

function filas(nombre) {
  const h = hoja(nombre);
  const v = h.getDataRange().getValues();
  const cab = v.shift();
  return v.filter(function (f) { return String(f[0]).trim() !== ''; })
          .map(function (f) {
            const o = {};
            cab.forEach(function (c, i) { o[String(c).trim()] = f[i]; });
            return o;
          });
}

function ahora() {
  return Utilities.formatDate(new Date(), husos(), 'yyyy-MM-dd HH:mm');
}

function husos() {
  return SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone() || 'Europe/Madrid';
}

/** '2026-09-19' -> 'Sábado 19 de septiembre de 2026' */
function fechaLarga(iso) {
  if (!iso) return '';
  const p = String(iso).slice(0, 10).split('-');
  const d = new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2]));
  if (isNaN(d.getTime())) return String(iso);
  const dia = DIAS[d.getDay()];
  return dia.charAt(0).toUpperCase() + dia.slice(1) +
         ' ' + d.getDate() + ' de ' + MESES[d.getMonth()] + ' de ' + d.getFullYear();
}

function texto(v) {
  if (v instanceof Date) return Utilities.formatDate(v, husos(), 'yyyy-MM-dd');
  return v === null || v === undefined ? '' : String(v).trim();
}

/* -------------------------------------------------------------- lectura */

function asegurarColumnasPartidos(h) {
  if (!h) return;
  const v = h.getDataRange().getValues();
  if (!v || !v.length) return;
  const cab = v[0].map(function(c) { return String(c).trim(); });
  const columnasExtra = ['Jornada', 'Rival', 'Ubicacion', 'EstadoPartido', 'MotivoAplazado'];
  const faltantes = [];
  columnasExtra.forEach(function(col) {
    if (cab.indexOf(col) === -1) faltantes.push(col);
  });
  if (faltantes.length > 0) {
    const colInicio = cab.length + 1;
    h.getRange(1, colInicio, 1, faltantes.length).setValues([faltantes]).setFontWeight('bold');
  }
}

function asegurarColumnasRespuestas(h) {
  if (!h) return;
  const v = h.getDataRange().getValues();
  if (!v || !v.length) return;
  const cab = v[0].map(function(c) { return String(c).trim(); });
  if (cab.indexOf('Desconvocado') === -1) {
    h.getRange(1, cab.length + 1).setValue('Desconvocado').setFontWeight('bold');
  }
}

function siguienteJornadaSugerida() {
  const ps = filas(HOJA_P);
  let maxJornada = 0;
  ps.forEach(function(p) {
    const num = parseInt(p.Jornada, 10);
    if (!isNaN(num) && num > maxJornada) maxJornada = num;
  });
  if (maxJornada > 0 && maxJornada < CALENDARIO_OFICIAL.length) {
    return maxJornada + 1;
  }
  const hoyStr = Utilities.formatDate(new Date(), husos(), 'yyyy-MM-dd');
  const proxima = CALENDARIO_OFICIAL.find(function(item) {
    return item.fecha >= hoyStr;
  });
  return proxima ? proxima.jornada : 1;
}

function partidoActivo() {
  const ps = filas(HOJA_P);
  const activos = ps.filter(function (p) { return texto(p.Estado) === 'activo'; });
  return activos.length ? activos[activos.length - 1] : null;
}

function convocatoria() {
  const hP = hoja(HOJA_P);
  if (hP) asegurarColumnasPartidos(hP);
  const hR = hoja(HOJA_R);
  if (hR) asegurarColumnasRespuestas(hR);

  const p = partidoActivo();

  const todos = filas(HOJA_J).map(function(j) {
    const activoTexto = String(j.Activo).trim().toLowerCase();
    const activo = activoTexto !== 'no' && activoTexto !== 'false' && activoTexto !== '0';

    return {
      id: texto(j.ID),
      nombre: texto(j.Nombre),
      dorsal: texto(j.Dorsal),
      activo: activo
    };
  });

  const jugadoresActivos = todos.filter(function(j) {
    return j.activo;
  });

  const rs = {};

  if (p) {
    filas(HOJA_R).forEach(function(r) {
      if (texto(r.IDPartido) === texto(p.ID)) {
        rs[texto(r.IDJugador)] = r;
      }
    });
  }

  const conRespuesta = function(j) {
    const r = rs[j.id] || {};

    return {
      id: j.id,
      nombre: j.nombre,
      dorsal: j.dorsal,
      activo: j.activo,
      estado: texto(r.Estado) || 'PENDIENTE',
      nota: texto(r.Nota),
      modificado: texto(r.Modificado),
      desconvocado: texto(r.Desconvocado) === 'si'
    };
  };

  const jugadores = jugadoresActivos
    .map(conRespuesta)
    .sort(function(a, b) {
      return a.nombre.localeCompare(b.nombre, 'es');
    });

  const jugadoresAdmin = todos
    .map(conRespuesta)
    .sort(function(a, b) {
      return a.nombre.localeCompare(b.nombre, 'es');
    });

  const cierre = p ? texto(p.Cierre) : '';

  return {
    hay: !!p,
    partido: p ? {
      id: texto(p.ID),
      fecha: texto(p.Fecha),
      fechaLarga: fechaLarga(texto(p.Fecha)),
      hora: texto(p.HoraPartido),
      llegada: texto(p.HoraLlegada),
      campo: texto(p.Campo),
      mapa: texto(p.Mapa),
      camiseta: texto(p.Camiseta),
      notas: texto(p.Notas),
      cierre: cierre,
      jornada: texto(p.Jornada),
      rival: texto(p.Rival),
      ubicacion: (texto(p.Ubicacion) || 'CASA').toUpperCase(),
      estadoPartido: (texto(p.EstadoPartido) || 'programado').toLowerCase(),
      motivoAplazado: texto(p.MotivoAplazado)
    } : null,
    cerrada: !!cierre &&
      (new Date().getTime() > new Date(cierre.replace(' ', 'T')).getTime()),
    jugadores: jugadores,
    jugadoresAdmin: jugadoresAdmin,
    calendario: CALENDARIO_OFICIAL,
    configCasa: CONFIG_CASA,
    siguienteJornada: siguienteJornadaSugerida(),
    resumen: {
      total: jugadores.length,
      si: jugadores.filter(function(j) {
        return j.estado === 'SI';
      }).length,
      no: jugadores.filter(function(j) {
        return j.estado === 'NO';
      }).length,
      pendiente: jugadores.filter(function(j) {
        return j.estado === 'PENDIENTE';
      }).length
    }
  };
}

/* -------------------------------------------------------------- escritura */

function guardarRespuesta(idJugador, estado, nota) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const c = convocatoria();
    if (!c.hay) throw new Error('No hay ninguna convocatoria abierta.');
    if (c.cerrada) throw new Error('El plazo para confirmar ya está cerrado. Avisa al responsable.');
    if (estado !== 'SI' && estado !== 'NO') throw new Error('Respuesta no válida.');

    const h = hoja(HOJA_R);
    const v = h.getDataRange().getValues();
    const idP = c.partido.id;
    for (var i = 1; i < v.length; i++) {
      if (texto(v[i][0]) === idP && texto(v[i][1]) === texto(idJugador)) {
        h.getRange(i + 1, 3, 1, 3).setValues([[estado, texto(nota).slice(0, 120), ahora()]]);
        return convocatoria();
      }
    }
    h.appendRow([idP, texto(idJugador), estado, texto(nota).slice(0, 120), ahora(), ahora()]);
    return convocatoria();
  } finally {
    lock.releaseLock();
  }
}

function cambiarEstadoAdmin(clave, idJugador, estado, nota) {
  comprobar(clave);
  if (estado === 'PENDIENTE') {
    const c = convocatoria();
    const h = hoja(HOJA_R);
    const v = h.getDataRange().getValues();
    for (var i = v.length - 1; i >= 1; i--) {
      if (texto(v[i][0]) === c.partido.id && texto(v[i][1]) === texto(idJugador)) h.deleteRow(i + 1);
    }
    return convocatoria();
  }
  return guardarRespuestaForzada(idJugador, estado, nota);
}

/** Marca o desmarca a un jugador como "Desconvocado" (excluido por el mister) en la convocatoria activa.
 *  No altera su respuesta SI/NO/PENDIENTE, y esa convocatoria no cuenta en sus estadísticas de presencia. */
function marcarDesconvocado(clave, idJugador, desconvocado) {
  comprobar(clave);
  const c = convocatoria();
  if (!c.hay) throw new Error('No hay ninguna convocatoria abierta.');

  const h = hoja(HOJA_R);
  asegurarColumnasRespuestas(h);
  const v = h.getDataRange().getValues();
  const cab = v[0].map(function(x) { return String(x).trim(); });
  const colDesc = cab.indexOf('Desconvocado');
  const colMod = cab.indexOf('Modificado');
  const idP = c.partido.id;

  for (var i = 1; i < v.length; i++) {
    if (texto(v[i][0]) === idP && texto(v[i][1]) === texto(idJugador)) {
      h.getRange(i + 1, colDesc + 1).setValue(desconvocado ? 'si' : '');
      if (colMod !== -1) h.getRange(i + 1, colMod + 1).setValue(ahora());
      return convocatoria();
    }
  }

  const nuevaFila = cab.map(function(col) {
    if (col === 'IDPartido') return idP;
    if (col === 'IDJugador') return texto(idJugador);
    if (col === 'Desconvocado') return desconvocado ? 'si' : '';
    if (col === 'Creado' || col === 'Modificado') return ahora();
    return '';
  });
  h.appendRow(nuevaFila);
  return convocatoria();
}

function guardarRespuestaForzada(idJugador, estado, nota) {
  const c = convocatoria();
  const h = hoja(HOJA_R);
  const v = h.getDataRange().getValues();
  for (var i = 1; i < v.length; i++) {
    if (texto(v[i][0]) === c.partido.id && texto(v[i][1]) === texto(idJugador)) {
      h.getRange(i + 1, 3, 1, 3).setValues([[estado, texto(nota), ahora()]]);
      return convocatoria();
    }
  }
  h.appendRow([c.partido.id, texto(idJugador), estado, texto(nota), ahora(), ahora()]);
  return convocatoria();
}

function guardarPartido(clave, datos) {
  comprobar(clave);
  const h = hoja(HOJA_P);
  asegurarColumnasPartidos(h);
  const v = h.getDataRange().getValues();
  const cab = v[0].map(function(c) { return String(c).trim(); });

  for (var i = 1; i < v.length; i++) {
    if (texto(v[i][0]) === texto(datos.id)) {
      const fila = i + 1;
      const mapeo = {
        'Fecha': datos.fecha,
        'HoraPartido': datos.hora,
        'HoraLlegada': datos.llegada,
        'Campo': datos.campo,
        'Mapa': datos.mapa,
        'Camiseta': datos.camiseta,
        'Notas': datos.notas,
        'Cierre': datos.cierre,
        'Jornada': datos.jornada !== undefined ? datos.jornada : '',
        'Rival': datos.rival !== undefined ? datos.rival : '',
        'Ubicacion': datos.ubicacion !== undefined ? datos.ubicacion : '',
        'EstadoPartido': datos.estadoPartido || 'programado',
        'MotivoAplazado': datos.motivoAplazado !== undefined ? datos.motivoAplazado : ''
      };

      Object.keys(mapeo).forEach(function(col) {
        const cIdx = cab.indexOf(col);
        if (cIdx !== -1) {
          h.getRange(fila, cIdx + 1).setValue(mapeo[col]);
        }
      });
      return convocatoria();
    }
  }
  throw new Error('Partido no encontrado.');
}

/** Archiva la convocatoria actual y abre una nueva (todos vuelven a PENDIENTE). */
function nuevoPartido(clave, datos) {
  comprobar(clave);
  const h = hoja(HOJA_P);
  asegurarColumnasPartidos(h);
  const v = h.getDataRange().getValues();
  const cab = v[0].map(function(c) { return String(c).trim(); });
  const colEstado = cab.indexOf('Estado');

  var max = 0;
  for (var i = 1; i < v.length; i++) {
    max = Math.max(max, Number(v[i][0]) || 0);
    if (colEstado !== -1 && texto(v[i][colEstado]) === 'activo') {
      h.getRange(i + 1, colEstado + 1).setValue('archivado');
    }
  }

  const mapeo = {
    'ID': max + 1,
    'Fecha': datos.fecha,
    'HoraPartido': datos.hora,
    'HoraLlegada': datos.llegada,
    'Campo': datos.campo,
    'Mapa': datos.mapa,
    'Camiseta': datos.camiseta,
    'Notas': datos.notas,
    'Cierre': datos.cierre,
    'Estado': 'activo',
    'Jornada': datos.jornada !== undefined ? datos.jornada : '',
    'Rival': datos.rival !== undefined ? datos.rival : '',
    'Ubicacion': datos.ubicacion !== undefined ? datos.ubicacion : '',
    'EstadoPartido': datos.estadoPartido || 'programado',
    'MotivoAplazado': datos.motivoAplazado !== undefined ? datos.motivoAplazado : ''
  };

  const nuevaFila = cab.map(function(col) {
    return mapeo.hasOwnProperty(col) ? mapeo[col] : '';
  });

  h.appendRow(nuevaFila);
  return convocatoria();
}

/** Cambia el estado del partido (programado / aplazado) */
function cambiarEstadoPartido(clave, idPartido, nuevoEstado, motivo) {
  comprobar(clave);
  const h = hoja(HOJA_P);
  asegurarColumnasPartidos(h);
  const v = h.getDataRange().getValues();
  const cab = v[0].map(function(c) { return String(c).trim(); });
  const colEstadoP = cab.indexOf('EstadoPartido');
  const colMotivo = cab.indexOf('MotivoAplazado');

  for (var i = 1; i < v.length; i++) {
    if (texto(v[i][0]) === texto(idPartido)) {
      if (colEstadoP !== -1) h.getRange(i + 1, colEstadoP + 1).setValue(nuevoEstado);
      if (colMotivo !== -1 && motivo !== undefined) h.getRange(i + 1, colMotivo + 1).setValue(motivo);
      return convocatoria();
    }
  }
  throw new Error('Partido no encontrado.');
}

function historial(clave) {
  comprobar(clave);
  const rs = filas(HOJA_R);
  return filas(HOJA_P).map(function (p) {
    const r = rs.filter(function (x) { return texto(x.IDPartido) === texto(p.ID); });
    return {
      id: texto(p.ID),
      fecha: texto(p.Fecha),
      fechaLarga: fechaLarga(texto(p.Fecha)),
      campo: texto(p.Campo),
      estado: texto(p.Estado),
      jornada: texto(p.Jornada),
      rival: texto(p.Rival),
      ubicacion: (texto(p.Ubicacion) || '').toUpperCase(),
      estadoPartido: (texto(p.EstadoPartido) || 'programado').toLowerCase(),
      motivoAplazado: texto(p.MotivoAplazado),
      si: r.filter(function (x) { return texto(x.Estado) === 'SI'; }).length,
      no: r.filter(function (x) { return texto(x.Estado) === 'NO'; }).length
    };
  }).reverse();
}

function estadisticas(clave) {
  comprobar(clave);

  const hR = hoja(HOJA_R);
  if (hR) asegurarColumnasRespuestas(hR);

  const jugadores = filas(HOJA_J);
  const partidos = filas(HOJA_P);
  const respuestas = filas(HOJA_R);

  // Las partidas aplazadas NO se contabilizan en las estadísticas de presencia
  const partidosContabilizados = partidos.filter(function(p) {
    const estado = texto(p.Estado);
    const estadoPartido = (texto(p.EstadoPartido) || 'programado').toLowerCase();
    return (estado === 'archivado' || estado === 'activo') && estadoPartido !== 'aplazado';
  });

  const porJugador = {};

  jugadores.forEach(function(j) {
    porJugador[texto(j.ID)] = {
      id: texto(j.ID),
      nombre: texto(j.Nombre),
      dorsal: texto(j.Dorsal),
      convocatorias: 0,
      si: 0,
      no: 0,
      pendiente: 0,
      desconvocado: 0,
      porcentaje: 0
    };
  });

  partidosContabilizados.forEach(function(p) {
    const idPartido = texto(p.ID);

    jugadores.forEach(function(j) {
      const idJugador = texto(j.ID);
      const dato = porJugador[idJugador];

      if (!dato) return;

      const respuesta = respuestas.find(function(r) {
        return texto(r.IDPartido) === idPartido &&
               texto(r.IDJugador) === idJugador;
      });

      // Si el mister lo desconvocó para ese partido, no cuenta en su estadística de presencia
      if (respuesta && texto(respuesta.Desconvocado) === 'si') {
        dato.desconvocado++;
        return;
      }

      dato.convocatorias++;

      if (!respuesta) {
        dato.pendiente++;
      } else if (texto(respuesta.Estado) === 'SI') {
        dato.si++;
      } else if (texto(respuesta.Estado) === 'NO') {
        dato.no++;
      } else {
        dato.pendiente++;
      }
    });
  });

  return {
    partidos: partidosContabilizados.length,
    jugadores: Object.keys(porJugador)
      .map(function(id) {
        const j = porJugador[id];
        j.porcentaje = j.convocatorias
          ? Math.round((j.si / j.convocatorias) * 100)
          : 0;
        return j;
      })
      .sort(function(a, b) {
        if (b.porcentaje !== a.porcentaje) return b.porcentaje - a.porcentaje;
        return b.si - a.si;
      })
  };
}

/** Devuelve la lista de jugadores y respuestas de un partido archivado/pasado */
function respuestasPartidoHistorial(clave, idPartido) {
  comprobar(clave);
  const p = filas(HOJA_P).find(function(x) { return texto(x.ID) === texto(idPartido); });
  if (!p) throw new Error('Partido no encontrado.');
  const jugadores = filas(HOJA_J);
  const rs = filas(HOJA_R).filter(function(r) { return texto(r.IDPartido) === texto(idPartido); });
  const mapR = {};
  rs.forEach(function(r) { mapR[texto(r.IDJugador)] = r; });

  return {
    partido: {
      id: texto(p.ID),
      jornada: texto(p.Jornada),
      rival: texto(p.Rival),
      fechaLarga: fechaLarga(texto(p.Fecha)),
      campo: texto(p.Campo),
      estado: texto(p.Estado),
      estadoPartido: (texto(p.EstadoPartido) || 'programado').toLowerCase()
    },
    jugadores: jugadores.map(function(j) {
      const r = mapR[texto(j.ID)] || {};
      return {
        id: texto(j.ID),
        nombre: texto(j.Nombre),
        dorsal: texto(j.Dorsal),
        estado: texto(r.Estado) || 'PENDIENTE',
        nota: texto(r.Nota)
      };
    }).sort(function(a, b) { return a.nombre.localeCompare(b.nombre, 'es'); })
  };
}

/** Permite al admin corregir la presencia (SI, NO, PENDIENTE) de un partido pasado */
function cambiarPresenciaHistorial(clave, idPartido, idJugador, nuevoEstado) {
  comprobar(clave);
  const h = hoja(HOJA_R);
  const v = h.getDataRange().getValues();
  const idP = texto(idPartido);
  const idJ = texto(idJugador);

  if (nuevoEstado === 'PENDIENTE') {
    for (var i = v.length - 1; i >= 1; i--) {
      if (texto(v[i][0]) === idP && texto(v[i][1]) === idJ) {
        h.deleteRow(i + 1);
        break;
      }
    }
  } else {
    var encontrado = false;
    for (var i = 1; i < v.length; i++) {
      if (texto(v[i][0]) === idP && texto(v[i][1]) === idJ) {
        h.getRange(i + 1, 3, 1, 3).setValues([[nuevoEstado, '', ahora()]]);
        encontrado = true;
        break;
      }
    }
    if (!encontrado) {
      h.appendRow([idP, idJ, nuevoEstado, '', ahora(), ahora()]);
    }
  }

  return respuestasPartidoHistorial(clave, idPartido);
}

function guardarJugador(clave, datos) {
  comprobar(clave);

  const id = texto(datos.id);
  const nombre = texto(datos.nombre);
  const dorsal = texto(datos.dorsal);
  const activo = datos.activo ? 'si' : 'no';

  if (!id) {
    throw new Error('Falta el ID del jugador.');
  }

  if (!nombre) {
    throw new Error('El nombre del jugador es obligatorio.');
  }

  const h = hoja(HOJA_J);
  const v = h.getDataRange().getValues();

  for (let i = 1; i < v.length; i++) {
    if (texto(v[i][0]) === id) {
      h.getRange(i + 1, 2, 1, 3).setValues([[
        nombre,
        dorsal,
        activo
      ]]);

      return convocatoria();
    }
  }

  throw new Error('Jugador no encontrado.');
}

function nuevoJugador(clave, datos) {
  comprobar(clave);

  const nombre = texto(datos.nombre);
  const dorsal = texto(datos.dorsal);

  if (!nombre) {
    throw new Error('El nombre del jugador es obligatorio.');
  }

  const h = hoja(HOJA_J);
  const v = h.getDataRange().getValues();

  let idMaximo = 0;

  for (let i = 1; i < v.length; i++) {
    idMaximo = Math.max(idMaximo, Number(v[i][0]) || 0);
  }

  h.appendRow([idMaximo + 1, nombre, dorsal, 'si']);

  return convocatoria();
}

function eliminarJugador(clave, idJugador) {
  comprobar(clave);

  const id = texto(idJugador);
  const h = hoja(HOJA_J);
  const v = h.getDataRange().getValues();

  const tieneHistorial = filas(HOJA_R).some(function(r) {
    return texto(r.IDJugador) === id;
  });

  if (tieneHistorial) {
    throw new Error(
      'No se puede eliminar este jugador porque tiene respuestas registradas. ' +
      'Desactívalo para conservar el historial.'
    );
  }

  for (let i = 1; i < v.length; i++) {
    if (texto(v[i][0]) === id) {
      h.deleteRow(i + 1);
      return convocatoria();
    }
  }

  throw new Error('Jugador no encontrado.');
}

function doPost(e) {
  let salida;
  try {
    const r = JSON.parse(e.postData.contents);
    const f = API_FUNZIONI[r.fn];
    if (!f) throw new Error('Funzione non valida');
    const data = f.apply(null, r.args || []);
    salida = { data: data === undefined ? null : data };
  } catch (err) {
    salida = { error: err.message };
  }
  return ContentService.createTextOutput(JSON.stringify(salida))
    .setMimeType(ContentService.MimeType.JSON);
}
